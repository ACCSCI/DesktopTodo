cmake_minimum_required(VERSION 3.24)
project(DesktopTodoNim LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(DEPS "${ROOT}/.tools/nim-duilib")
set(DUI "${DEPS}/source")
set(DUILIB_ENABLE_SDL OFF CACHE BOOL "" FORCE)
set(DUILIB_ENABLE_CEF OFF CACHE BOOL "" FORCE)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN DUILIB_CEF=0 DUILIB_SDL=0)
add_compile_options(/utf-8)
add_subdirectory("${DUI}/duilib/third_party/zlib" zlib)
set(ZLIB_INCLUDE_DIR "${DUI}/duilib/third_party/zlib" CACHE PATH "" FORCE)
set(ZLIB_LIBRARY zlibstatic CACHE STRING "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
add_subdirectory("${DUI}/duilib/third_party/libpng" png)
set_property(TARGET png_static PROPERTY LINK_LIBRARIES zlibstatic)
set_property(TARGET png_static PROPERTY INTERFACE_LINK_LIBRARIES zlibstatic)
add_subdirectory("${DUI}/duilib/third_party/libwebp" webp)
add_subdirectory("${DUI}/duilib/third_party/cximage" cximage)
add_subdirectory("${DUI}/duilib" duilib)
# The pinned upstream CMake enables WebView2 unconditionally. This application
# uses only native controls, and excludes both browser integrations explicitly.
get_target_property(DUI_DEFS duilib COMPILE_DEFINITIONS)
list(FILTER DUI_DEFS EXCLUDE REGEX "^DUILIB_WEBVIEW2=")
set_property(TARGET duilib PROPERTY COMPILE_DEFINITIONS "${DUI_DEFS};DUILIB_WEBVIEW2=0")
get_target_property(DUI_SOURCES duilib SOURCES)
list(FILTER DUI_SOURCES EXCLUDE REGEX "/(CEFControl|WebView2)/")
set_property(TARGET duilib PROPERTY SOURCES "${DUI_SOURCES}")
target_compile_definitions(duilib PUBLIC DUILIB_WEBVIEW2=0 DUILIB_CEF=0 DUILIB_SDL=0 SK_GANESH SK_GL SK_RELEASE)
target_include_directories(duilib PUBLIC "${DUI}" "${DEPS}/skia")
file(GLOB_RECURSE PRODUCT CONFIGURE_DEPENDS "${ROOT}/app/src/*.cpp")
list(FILTER PRODUCT EXCLUDE REGEX "/editor.cpp$")
file(GLOB FRONTEND CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/src/*.cpp")
add_library(todo_sqlite STATIC "${ROOT}/.tools/sqlite/sqlite3.c")
target_compile_definitions(todo_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
add_executable(DesktopTodoNim WIN32 ${PRODUCT} ${FRONTEND} "${ROOT}/app/resources/app.rc")
target_compile_definitions(DesktopTodoNim PRIVATE TODO_USE_NIM=1)
target_include_directories(DesktopTodoNim PRIVATE "${ROOT}/app/include" "${ROOT}/.tools/sqlite" "${ROOT}/app/resources")
target_link_directories(DesktopTodoNim PRIVATE "${DEPS}/skia/out/llvm.x64.release")
target_link_libraries(DesktopTodoNim PRIVATE duilib duilib-cximage duilib-webp png_static zlibstatic
  svg skshaper skottie sksg jsonreader skia todo_sqlite
  user32 gdi32 comctl32 shell32 ole32 oleaut32 advapi32 comdlg32 shlwapi dwmapi imm32
  dwrite dxgi d3d11 opengl32 glu32 winmm msimg32 usp10 windowscodecs uuid version uxtheme)
set_target_properties(DesktopTodoNim PROPERTIES OUTPUT_NAME "DesktopTodo-Nim" RUNTIME_OUTPUT_DIRECTORY "${ROOT}/.tools/nim-duilib/bin")
target_link_options(DesktopTodoNim PRIVATE /MANIFEST:NO)
target_link_options(DesktopTodoNim PRIVATE "/MAP:${DEPS}/build/DesktopTodo-Nim.map")
